# ---- Evaluation harness -----------------------------------------------------
# !! MTEB v2 IS REQUIRED - NOT v1. !!
# The pipeline is written against the v2 API surface:
#   - `mteb.evaluate(model, tasks=...)`     (v1 used `MTEB(tasks).run(model)`)
#   - `AbsEncoder`                          (v2 `encode()` takes a DataLoader of
#                                            batch dicts, NOT a `list[str]`)
#   - `SearchProtocol`                      (v2-only: custom `index()`/`search()`
#                                            so our real hybrid pipeline - not a
#                                            bare encoder - produces the scores)
mteb==2.20.11         # benchmark harness; provides AppsRetrieval + NDCG/MRR

# ---- Embedding models -------------------------------------------------------
sentence-transformers==6.0.1  # bi-encoder + cross-encoder reranker checkpoints

# transformers is imported DIRECTLY by shipped code (src/config.py resolves a
# checkpoint's context window through AutoTokenizer), so it must be declared
# here rather than inherited. Until now it resolved only transitively via
# sentence-transformers, which means a future sentence-transformers release
# could move or drop that pin underneath us - and the failure would surface as
# a wrong context window, i.e. quietly worse retrieval, not an ImportError.
# Pinned to the version this stack already resolves and every number in
# experiments.md was measured with.
transformers==5.17.0  # AutoTokenizer, used directly by src/config.py

# ---- Lexical retrieval ------------------------------------------------------
# rank-bm25 MOVED to requirements-dev.txt: nothing imports it. The BM25
# diagnostic implements Okapi directly over a scipy sparse matrix because
# rank_bm25 measured at ~0.6 s/query here (see scripts/bm25_diagnostic.py),
# and BM25 is a gated-out stage in any case.

# ---- Vector index -----------------------------------------------------------
faiss-cpu==1.15.0     # ANN index over snippet embeddings; CPU build only

# ---- Data loading -----------------------------------------------------------
datasets==5.0.1       # pulls CoIR-Retrieval/apps corpus + queries + qrels
huggingface_hub==1.31.0  # model/dataset download + local cache management

# ---- Numerics / UX ----------------------------------------------------------
numpy==2.4.6          # embedding matrices, score fusion arithmetic
tqdm==4.70.1          # progress bars for the long encode/index passes

# ---- Testing ----------------------------------------------------------------
# pytest MOVED to requirements-dev.txt: it is a test dependency, not a
# runtime one. The image installs both, so the repro gate still runs.
